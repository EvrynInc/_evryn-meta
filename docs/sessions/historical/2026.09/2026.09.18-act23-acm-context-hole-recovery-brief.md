```js
// READ-ONLY. Prints one dead branch of a Claude Code transcript, oldest first, by walking parentUuid
// back from the branch's LEAF uuid until it reaches the FORK uuid (the on-chain message it hangs off).
// Usage: node branch-extract.mjs <transcript.jsonl> <leafUuid> <forkUuid> [maxResultChars=1500]
import fs from 'fs';
const [file, leaf, fork, maxArg] = process.argv.slice(2);
const MAX = Number(maxArg || 1500);
const byUuid = new Map();
for (const line of fs.readFileSync(file, 'utf8').split('\n')) {
  let o; try { o = JSON.parse(line); } catch { continue; }
  if (o.uuid && !byUuid.has(o.uuid)) byUuid.set(o.uuid, o);
}
const chain = []; let u = leaf;
while (u && u !== fork && byUuid.has(u)) { chain.push(byUuid.get(u)); u = byUuid.get(u).parentUuid; }
if (u !== fork) console.log(`WARNING: walked back to ${u || 'null'} without meeting the fork ${fork} - check the uuids.`);
chain.reverse();
const pt = t => (t ? new Date(new Date(t).getTime() - 7 * 3600e3).toISOString().slice(5, 19).replace('T', ' ') + ' PT' : '--');
const cut = s => (s.length > MAX ? s.slice(0, MAX) + ` [...${s.length - MAX} more chars]` : s);
for (const o of chain) {
  const c = o.message && o.message.content;
  if (o.type !== 'user' && o.type !== 'assistant') continue;
  const parts = typeof c === 'string' ? [{ type: 'text', text: c }] : Array.isArray(c) ? c : [];
  for (const p of parts) {
    if (p.type === 'text' && p.text.trim()) console.log(`\n[${pt(o.timestamp)}] ${o.type.toUpperCase()} TEXT:\n${cut(p.text)}`);
    else if (p.type === 'tool_use') console.log(`\n[${pt(o.timestamp)}] TOOL CALL ${p.name}: ${cut(JSON.stringify(p.input))}`);
    else if (p.type === 'tool_result') {
      const r = typeof p.content === 'string' ? p.content : JSON.stringify(p.content);
      console.log(`[${pt(o.timestamp)}] TOOL RESULT${p.is_error ? ' (ERROR)' : ''}: ${cut(r)}`);
    }
  }
}
console.log(`\n(${chain.length} messages on this branch)`);
```

⚠️ **It carries a WARNING line if the walk does not meet the fork — trust the output only when that line is absent.** **Timestamps print in PT (UTC−7).**
